  const state = {materials:[],questions:[],attempts:[],vocabulary:[],activeMaterial:null,quiz:[],answers:{},quizMode:'Practice',startedAt:0,timer:null,secondsLeft:0};

  const esc = s => String(s == null ? '' : s).replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));

  function notice(message, error=false){const el=$('globalNotice');el.textContent=message;el.classList.remove('hidden','error');if(error)el.classList.add('error');window.scrollTo({top:0,behavior:'smooth'});setTimeout(()=>el.classList.add('hidden'),9000)}

  function callServer(name,args,onSuccess,onFailure){

    const runner=google.script.run

      .withSuccessHandler(onSuccess||(()=>{}))

      .withFailureHandler(e=>{

        if(typeof onFailure==='function'){onFailure(e);return}

        notice(e&&e.message?e.message:String(e),true)

      });

    runner[name](...(args||[]))

  }

  function showPage(page){document.querySelectorAll('.page').forEach(p=>p.classList.toggle('active',p.id==='page-'+page));document.querySelectorAll('#nav button').forEach(b=>b.classList.toggle('active',b.dataset.page===page));const labels={dashboard:'Dashboard',reviewers:'My Reviewers','upload':'Upload & Scan',quiz:'Practice Quiz',mock:'Mock Exam',dictionary:'Dictionary',progress:'My Progress',saved:'Saved Questions'};$('pageTitle').textContent=labels[page]||'Reviewer';if(page==='reviewers')renderMaterials();if(page==='saved')renderQuestions();if(page==='dictionary')renderVocabulary();if(page==='progress')renderProgress();if(page==='dashboard')renderDashboard();if(page==='quiz'||page==='mock'){}}

  function renderDashboard(){$('statMaterials').textContent=state.materials.length;$('statQuestions').textContent=state.questions.length;$('statAttempts').textContent=state.attempts.length;const list=$('recentAttempts');list.innerHTML=state.attempts.slice(0,5).map(a=>`<div class="item"><div class="item-head"><div><h3>${esc(a.mode)} · ${esc(a.subject)}</h3><div class="muted small">${esc(new Date(a.createdAt).toLocaleString())}</div></div><span class="pill">${Number(a.total)?Math.round(Number(a.score)/Number(a.total)*100):0}%</span></div><div class="small">${esc(a.score)} / ${esc(a.total)} correct</div></div>`).join('')||'<div class="empty">No attempts yet. Start a quiz when your question bank is ready.</div>'}

  function renderMaterials(){const list=$('materialsList');list.innerHTML=state.materials.map(m=>`<div class="item"><div class="item-head"><div><h3>${esc(m.title||m.fileName)}</h3><div class="muted small">${esc(m.subject)}${m.topic?' · '+esc(m.topic):''}</div><div class="muted small">${esc(m.fileName||'Manual text')} · ${String(m.extractedText||'').length} characters</div></div><button class="btn secondary" onclick="editMaterial('${esc(m.id)}')">Open / edit</button></div></div>`).join('')||'<div class="empty">No reviewers yet. Upload a file or paste your notes.</div>'}

  function saveManualMaterial(){const title=$('manualTitle').value.trim(),text=$('manualText').value.trim();if(!title||!text){notice('Enter a title and reviewer text.',true);return}const m={id:'manual-'+Date.now(),title,subject:$('manualSubject').value,topic:$('manualTopic').value,extractedText:text,fileName:'Manual entry',mimeType:'text/plain',driveFileId:'',notes:'Manually entered'};state.materials.unshift(m); // Use backend upload-free endpoint below

    callServer('saveManualMaterial',[m],saved=>{state.materials=state.materials.filter(x=>x.id!==m.id);state.materials.unshift(saved);renderMaterials();renderDashboard();$('manualTitle').value='';$('manualTopic').value='';$('manualText').value='';notice('Reviewer saved.')})}

  function editMaterial(id){const m=state.materials.find(x=>String(x.id)===String(id));if(!m)return;state.activeMaterial=m;$('materialEditor').classList.remove('hidden');$('editMaterialId').value=m.id;$('editMaterialTitle').value=m.title||'';$('editMaterialSubject').value=m.subject||'';$('editMaterialTopic').value=m.topic||'';$('editMaterialText').value=m.extractedText||'';$('generateStatus').innerHTML='';$('materialEditor').scrollIntoView({behavior:'smooth'})}

  function saveMaterialEdit(){const p={id:$('editMaterialId').value,title:$('editMaterialTitle').value,subject:$('editMaterialSubject').value,topic:$('editMaterialTopic').value,extractedText:$('editMaterialText').value,notes:''};callServer('saveMaterial',[p],()=>{Object.assign(state.materials.find(m=>String(m.id)===String(p.id)),p);renderMaterials();notice('Reviewer updated.')})}

  function removeMaterial(){const id=$('editMaterialId').value;if(!confirm('Delete this reviewer and its uploaded Drive file?'))return;callServer('deleteMaterial',[id],()=>{state.materials=state.materials.filter(m=>String(m.id)!==String(id));$('materialEditor').classList.add('hidden');renderMaterials();renderDashboard();notice('Reviewer deleted.')})}

  function generateForMaterial(){const id=$('editMaterialId').value;$('generateStatus').innerHTML='<div class="notice">Generating questions from the text. Please wait…</div>';callServer('generateQuestionsFromMaterial',[id,10,'Mixed'],res=>{state.questions=state.questions.concat(res.questions||[]);renderQuestions();renderDashboard();$('generateStatus').innerHTML=`<div class="notice">Created ${res.count} questions. Review them in Saved Questions.</div>`})}

  $('uploadForm').addEventListener('submit',e=>{

    e.preventDefault();

    const file=$('uploadFile').files[0];

    if(!file)return;

    const btn=$('uploadBtn');

    btn.disabled=true;

    btn.textContent='Uploading…';

    $('uploadStatus').innerHTML='<div class="notice">Uploading file and extracting text. Please wait…</div>';

    const reader=new FileReader();

    reader.onload=()=>{

      const base64=reader.result.split(',')[1];

      const payload={fileName:file.name,mimeType:file.type||'application/octet-stream',base64,title:$('uploadTitle').value.trim(),subject:$('uploadSubject').value,topic:$('uploadTopic').value.trim()};

      callServer('uploadMaterial',[payload],res=>{

        btn.disabled=false;

        btn.textContent='Upload and scan';

        if(res && res.material){

          state.materials=state.materials.filter(m=>String(m.id)!==String(res.material.id));

          state.materials.unshift(res.material);

          renderMaterials();

          renderDashboard();

          $('uploadStatus').innerHTML=`<div class="notice">${esc(res.extractionMessage||'Upload complete. Open My Reviewers to check extracted text.')}</div><button class="btn secondary" onclick="showPage('reviewers');editMaterial('${esc(res.material.id)}')">Review extracted text</button>`;

          $('uploadForm').reset();

        } else {

          $('uploadStatus').innerHTML='<div class="notice error">The server returned no material record. Refresh and check My Reviewers.</div>';

          refreshData();

        }

      },err=>{

        btn.disabled=false;

        btn.textContent='Upload and scan';

        const message=err&&err.message?err.message:String(err||'Unknown error');

        $('uploadStatus').innerHTML=`<div class="notice error">The request failed: ${esc(message)}<br><br>The file may already be saved. Refresh and check My Reviewers before uploading it again.</div>`;

        refreshData();

      });

    };

    reader.onerror=()=>{

      btn.disabled=false;

      btn.textContent='Upload and scan';

      $('uploadStatus').innerHTML='<div class="notice error">Could not read the selected file. Please choose it again.</div>';

    };

    reader.readAsDataURL(file);

  });

  function renderQuestions(){const list=$('questionsList');list.innerHTML=state.questions.map(q=>`<div class="item"><div class="item-head"><div><h3>${esc(q.question)}</h3><div class="muted small">${esc(q.subject)} · ${esc(q.topic)} · ${esc(q.difficulty)}</div><div class="small">Answer: ${esc(q.correctAnswer)} — ${esc(q.explanation||'No explanation added')}</div></div><div class="actions"><button class="btn secondary" onclick="editQuestion('${esc(q.id)}')">Edit</button><button class="btn danger" onclick="removeQuestion('${esc(q.id)}')">Delete</button></div></div></div>`).join('')||'<div class="empty">No questions saved yet. Add a question manually or generate questions from a reviewer.</div>'}

  function openQuestionForm(){['qId','qTopic','qQuestion','qA','qB','qC','qD','qExplanation'].forEach(id=>$(id).value='');$('questionFormTitle').textContent='Add question';$('questionForm').classList.remove('hidden');$('questionForm').scrollIntoView({behavior:'smooth'})}

  function editQuestion(id){const q=state.questions.find(x=>String(x.id)===String(id));if(!q)return;$('qId').value=q.id;$('qSubject').value=q.subject;$('qDifficulty').value=q.difficulty;$('qTopic').value=q.topic;$('qQuestion').value=q.question;$('qA').value=q.optionA;$('qB').value=q.optionB;$('qC').value=q.optionC;$('qD').value=q.optionD;$('qCorrect').value=q.correctAnswer;$('qExplanation').value=q.explanation;$('questionFormTitle').textContent='Edit question';$('questionForm').classList.remove('hidden');$('questionForm').scrollIntoView({behavior:'smooth'})}

  function saveQuestionForm(){const q={id:$('qId').value||undefined,subject:$('qSubject').value,difficulty:$('qDifficulty').value,topic:$('qTopic').value,question:$('qQuestion').value,optionA:$('qA').value,optionB:$('qB').value,optionC:$('qC').value,optionD:$('qD').value,correctAnswer:$('qCorrect').value,explanation:$('qExplanation').value};if(!q.question.trim()){notice('Enter a question.',true);return}callServer('saveQuestion',[q],saved=>{if(q.id)state.questions=state.questions.map(x=>String(x.id)===String(q.id)?saved:x);else state.questions.push(saved);renderQuestions();renderDashboard();$('questionForm').classList.add('hidden');notice('Question saved.')})}

  function removeQuestion(id){if(!confirm('Delete this question?'))return;callServer('deleteQuestion',[id],()=>{state.questions=state.questions.filter(q=>String(q.id)!==String(id));renderQuestions();renderDashboard();notice('Question deleted.')})}

  function startQuiz(mode){const subject=$('quizSubject').value;let count=Number(mode==='Mock Exam'?$('mockCount').value:$('quizCount').value);let pool=state.questions.filter(q=>subject==='All Subjects'||q.subject===subject);if(!pool.length){notice('No questions match this selection. Add questions or generate them from your uploaded reviewers.',true);return}pool=pool.sort(()=>Math.random()-.5).slice(0,count);state.quiz=pool;state.answers={};state.quizMode=mode;state.startedAt=Date.now();state.secondsLeft=mode==='Mock Exam'?Number($('mockMinutes').value)*60:0;renderQuizArea(mode);if(mode==='Mock Exam')startTimer();showPage(mode==='Mock Exam'?'mock':'quiz')}

  function renderQuizArea(mode){const area=$(mode==='Mock Exam'?'mockArea':'quizArea');area.classList.remove('hidden');area.innerHTML=`<div class="item-head"><div><h2>${esc(mode)}</h2><div class="muted small">${state.quiz.length} questions</div></div><div class="timer" id="quizTimer">${mode==='Mock Exam'?formatTime(state.secondsLeft):'Practice mode'}</div></div><div id="quizQuestions"></div><button class="btn" onclick="submitQuiz()">Submit answers</button>`;const qWrap=area.querySelector('#quizQuestions');qWrap.innerHTML=state.quiz.map((q,i)=>`<div class="question" id="question-${i}"><strong>${i+1}. ${esc(q.question)}</strong>${['A','B','C','D'].map(letter=>`<label class="option" id="opt-${i}-${letter}"><input type="radio" name="answer-${i}" value="${letter}" onchange="state.answers[${i}]='${letter}'"><span><b>${letter}.</b> ${esc(q['option'+letter])}</span></label>`).join('')}<div id="explain-${i}" class="answer-explain hidden"></div></div>`).join('')}

  function formatTime(s){const m=Math.floor(Math.max(0,s)/60),sec=Math.max(0,s)%60;return String(m).padStart(2,'0')+':'+String(sec).padStart(2,'0')}

  function startTimer(){clearInterval(state.timer);state.timer=setInterval(()=>{state.secondsLeft--;const el=$('quizTimer');if(el)el.textContent=formatTime(state.secondsLeft);if(state.secondsLeft<=0){clearInterval(state.timer);state.timer=null;submitQuiz(true)}},1000)}

  function submitQuiz(auto=false){if(!state.quiz.length)return;clearInterval(state.timer);state.timer=null;let score=0;state.quiz.forEach((q,i)=>{const chosen=state.answers[i]||'';if(chosen===String(q.correctAnswer).toUpperCase())score++;['A','B','C','D'].forEach(letter=>{const el=$('opt-'+i+'-'+letter);if(!el)return;if(letter===String(q.correctAnswer).toUpperCase())el.classList.add('correct');else if(letter===chosen)el.classList.add('wrong');const radio=el.querySelector('input');if(radio)radio.disabled=true});const ex=$('explain-'+i);ex.classList.remove('hidden');ex.innerHTML=`<b>Correct answer: ${esc(q.correctAnswer)}</b><br>${esc(q.explanation||'No explanation was saved for this question.')}`});const duration=Math.round((Date.now()-state.startedAt)/1000);const payload={mode:state.quizMode,subject:$('quizSubject').value,score,total:state.quiz.length,durationSeconds:duration,answers:state.quiz.map((q,i)=>({questionId:q.id,chosen:state.answers[i]||'',correctAnswer:q.correctAnswer}))};callServer('saveAttempt',[payload],attempt=>{state.attempts.unshift(attempt);renderDashboard();renderProgress()});const area=$(state.quizMode==='Mock Exam'?'mockArea':'quizArea');const result=document.createElement('div');result.className='notice';result.innerHTML=`<b>${auto?'Time is up. ':''}Result: ${score} / ${state.quiz.length} (${Math.round(score/state.quiz.length*100)}%)</b><br>${score===state.quiz.length?'Excellent work!': 'Review the explanations and retry the questions you missed.'}`;area.prepend(result);area.querySelectorAll('button').forEach(b=>b.disabled=true);notice('Your result has been recorded.')}

  function renderProgress(){const total=state.attempts.length;const average=total?Math.round(state.attempts.reduce((s,a)=>s+(Number(a.total)?Number(a.score)/Number(a.total):0),0)/total*100):0;$('progressAttempts').textContent=total;$('progressAverage').textContent=average+'%';$('progressAnswered').textContent=state.attempts.reduce((s,a)=>s+Number(a.total||0),0);$('attemptList').innerHTML=state.attempts.map(a=>`<div class="item"><div class="item-head"><div><h3>${esc(a.mode)} · ${esc(a.subject)}</h3><div class="muted small">${esc(new Date(a.createdAt).toLocaleString())}</div></div><span class="pill">${Number(a.total)?Math.round(Number(a.score)/Number(a.total)*100):0}%</span></div><div class="small">${esc(a.score)} / ${esc(a.total)} · ${Math.round(Number(a.durationSeconds||0)/60)} min</div></div>`).join('')||'<div class="empty">No attempt history yet.</div>'}

  function lookupWord(){const word=$('dictWord').value.trim();if(!word){notice('Enter a word first.',true);return}$('dictResult').innerHTML='<div class="notice">Looking up the word…</div>';callServer('getDictionaryEntry',[word],entry=>{state.vocabulary.unshift(entry);renderVocabulary();$('dictResult').innerHTML=`<div class="item"><h3>${esc(entry.word)} ${entry.pronunciation?'· '+esc(entry.pronunciation):''}</h3><p>${esc(entry.definition)}</p><p><b>Example:</b> ${esc(entry.example)}</p><p><b>Synonyms:</b> ${esc(entry.synonyms)}</p><p><b>Antonyms:</b> ${esc(entry.antonyms||'No common antonym listed')}</p></div>`})}

  function renderVocabulary(){$('vocabList').innerHTML=state.vocabulary.map(v=>`<div class="item"><div class="item-head"><div><h3>${esc(v.word)} ${v.pronunciation?'· '+esc(v.pronunciation):''}</h3><div>${esc(v.definition)}</div><div class="muted small">Synonyms: ${esc(v.synonyms)} · Antonyms: ${esc(v.antonyms)}</div></div><button class="btn danger" onclick="removeVocabulary('${esc(v.id)}')">Delete</button></div></div>`).join('')||'<div class="empty">Your saved vocabulary will appear here.</div>'}

  function removeVocabulary(id){if(!confirm('Delete this vocabulary entry?'))return;callServer('deleteVocabulary',[id],()=>{state.vocabulary=state.vocabulary.filter(v=>String(v.id)!==String(id));renderVocabulary()})}
